void main() {
  List<String> alunos = ['vitor', 'lucas', 'davi', 'guilherme', 'laura'];


  print('$alunos');


  String procurado = 'davi';
  bool encontrou = false;


  for (int i = 0; i < alunos.length; i++) {
    if (alunos[i] == procurado) {
      encontrou = true;
      print('\n$procurado encontrado na posição $i.');
      break;
    }
  }


  if (!encontrou) {
    print('\n$alunos não foi encontrado.');
  }
}

  List<double> notas = [8.5, 7.0, 9.5, 6.0, 10.0];

  double media = soma / 5;


  print('A media é: $media');


if (media >= 7) {
    print('Aprovado!');
   } else {
    print('Reprovado!');
  }
}
